pt() {  # pt LOG NODEID...: one fresh pytest process on GPU 0
    local log=$1; shift
    (cd /tmp && CUDA_LAUNCH_BLOCKING=1 CUDA_VISIBLE_DEVICES=0 PYTHONPATH=$T/tests/unittest:$T/tests timeout -k 30 900 \
        python3 -m pytest -q -p no:cacheprovider --tb=short -rA --basetemp=/tmp/bt-ima "$@" > "$log" 2>&1)
    echo "rc=$? | $(grep -F -- "$ID" "$log" | grep -E '^(PASSED|FAILED|ERROR|SKIPPED)' | cut -c1-8 | tr '\n' ' ')| $(tail -1 "$log")"
}
arm() {  # arm NAME: collect, 3 isolated runs, 1 window run
    local a=$1 pre
    (cd /tmp && PYTHONPATH=$T/tests/unittest:$T/tests timeout 600 python3 -m pytest --collect-only -q -p no:cacheprovider \
        "$TF" 2>&1 | grep '::' > /tmp/ids-$a.txt)
    pre=$(python3 - /tmp/ids-$a.txt "$ID" "$TF" <<'PY'
import sys
ids = open(sys.argv[1]).read().split()
i = next((k for k, x in enumerate(ids) if x.endswith("::" + sys.argv[2])), None)
# Collected ids are relative to tests/unittest; pytest runs from /tmp, so give it the absolute file path.
print("COLLECT_FAIL" if i is None else " ".join(sys.argv[3] + "::" + x.split("::", 1)[1] for x in ids[max(0, i - 10):i + 1]))
PY
)
    echo "## arm $a: collected $(wc -l < /tmp/ids-$a.txt); window: $pre"
    [[ $pre == COLLECT_FAIL ]] && { echo "## arm $a INVALID (failing id not collected)"; return; }
    local fid=${pre##* }
    for r in 1 2 3; do echo "## arm $a iso$r $(pt $R/review/diag/moemodule-ima-$job-$a-iso$r.log "$fid")"; done
    echo "## arm $a window $(pt $R/review/diag/moemodule-ima-$job-$a-window.log $pre)"
    grep -hE '^(PASSED|FAILED|ERROR)' $R/review/diag/moemodule-ima-$job-$a-window.log | cut -c1-200
}
