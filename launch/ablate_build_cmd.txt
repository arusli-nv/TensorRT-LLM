disp() { echo "${ns}20moeA2ADispatchKernelILi8ELb$1ELb$2ELb0ELb0ELi$3EEEvPKiNS2_22DispatchKernelPointersEiiiiiii"; }

insns() { grep -E '^\s+/\*[0-9a-f]{4}\*/|^\s+/\* 0x[0-9a-f]{16} \*/' "$1"; }  # both encoding words
