# PROBE3 (1 node; DIAGNOSTIC, decisive): does compaction stall CPU readers by source memory type? scratch-probe3/probe3.c (C, one process): 16 readers copy 16 MiB chunks into driver-pinned dsts from S1 /dev/shm pre-faulted, S2 /dev/shm lazy (PTEs dropped per block), S3 cuMemCreate PINNED HOST_NUMA (60 GiB each, GPU0's node), interleaved in 5 s blocks S1,S2,S3,...; 45 s quiet then 150 s with a C fragmenter child forcing direct compaction. Output: review/diag/probe3-<job>.txt (PROBE3/PHASE/BLOCK/DSTATE lines). ~6 min; --time 00:30.
R=/lustre/fs1/portfolios/coreai/projects/coreai_comparch_trtllm/users/arusli/src/TensorRT-LLM/.wideep-ft-runs/claude
bash $R/tools/check_excluded_nodes.sh || exit 1
out=$R/review/diag/probe3-${SLURM_JOB_ID:-x}.txt
{
    uname -a; ulimit -l; nvidia-smi -L
    for f in /proc/sys/vm/compaction_proactiveness /proc/sys/vm/compact_unevictable_allowed /proc/sys/vm/extfrag_threshold \
             /sys/kernel/mm/transparent_hugepage/enabled /sys/kernel/mm/transparent_hugepage/defrag \
             /sys/kernel/mm/transparent_hugepage/shmem_enabled /sys/kernel/mm/transparent_hugepage/hpage_pmd_size; do
        echo "$f: $(cat $f 2>&1)"
    done
    df -h /dev/shm; free -g
    bin=$(mktemp -d)
    gcc -O2 -std=gnu11 -Wall -I/usr/local/cuda/include $R/scratch-probe3/probe3.c -o $bin/probe3 \
        -L/usr/local/cuda/lib64/stubs -lcuda -lpthread -lrt || { echo PROBE3_BUILD_FAILED; exit 1; }
    ldd $bin/probe3 | grep -i cuda
    cd $bin && timeout 900 ./probe3
    echo "probe3 rc=$?"
    free -g | head -2
    echo PROBE3_CMD_DONE
} > "$out" 2>&1
