pt() {  # pt LOG NODEID...: one fresh pytest process, all GPUs visible
    local log=$1 t0=$SECONDS; shift
    (cd /tmp && PYTHONPATH=$T/tests/unittest:$T/tests timeout -k 30 900 \
        python3 -m pytest -q -p no:cacheprovider --tb=short -rA --basetemp=/tmp/bt-prefix "$@" > "$log" 2>&1)
    local rc=$? el=$((SECONDS - t0))
    echo "rc=$rc wall=${el}s | $(grep -F -- "$ID" "$log" | grep -E '^(PASSED|FAILED|ERROR|SKIPPED)' | cut -c1-8 | tr '\n' ' ')| $(tail -1 "$log")"
}
arm() {  # arm NAME: collect, 2 runs of the ordered prefix
    local a=$1 pre
    (cd /tmp && PYTHONPATH=$T/tests/unittest:$T/tests timeout 600 python3 -m pytest --collect-only -q -p no:cacheprovider \
        "$TF" 2>&1 | grep '::' > /tmp/ids-$a.txt)
    pre=$(python3 - /tmp/ids-$a.txt "$ID" "$TF" <<'PY'
import sys
ids = open(sys.argv[1]).read().split()
i = next((k for k, x in enumerate(ids) if x.endswith("::" + sys.argv[2])), None)
# Collected ids are relative to tests/unittest; pytest runs from /tmp, so give it the absolute file path.
print("COLLECT_FAIL" if i is None else " ".join(sys.argv[3] + "::" + x.split("::", 1)[1] for x in ids[:i + 1]))
PY
)
    echo "## arm $a: collected $(wc -l < /tmp/ids-$a.txt); window: $pre"
    [[ $pre == COLLECT_FAIL ]] && { echo "## arm $a INVALID (failing id not collected)"; return; }
    echo "## arm $a prefix length $(echo $pre | wc -w)"
    for r in 1 2; do echo "## arm $a prefix$r $(pt $R/review/diag/moemodule-prefix-$job-$a-prefix$r.log $pre)"
        grep -hE '^(FAILED|ERROR)' $R/review/diag/moemodule-prefix-$job-$a-prefix$r.log | head -5 | cut -c1-250; done
}
